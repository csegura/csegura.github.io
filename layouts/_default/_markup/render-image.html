{{- /* Pictures in a post: lazy loading, the size of the picture (so the text does not jump), smaller copies for small screens, a caption when the alt text is a real sentence, and click to enlarge. */ -}} {{- $dest := .Destination -}} {{- $alt := .PlainText | default .Text -}} {{- $caption := "" -}} {{- if ge (len (findRE `\s+` $alt)) 3 }}{{ $caption = $alt }}{{ end -}} {{- if findRE `^[:;]-?[)(DPp]$` $alt -}} {{- else if findRE `(?i)\.mp4$` $dest -}} {{- else -}} {{- $res := partial "image-resource.html" $dest -}} {{- if $res -}} {{- $w := $res.Width -}} {{- $small := $res -}} {{- if gt $w 900 }}{{ $small = $res.Resize "800x webp q80" }}{{ end -}} {{- $big := $res -}} {{- if gt $w 1700 }}{{ $big = $res.Resize "1600x webp q82" }}{{ end -}} {{ with $caption }}{{ . }}{{ end }} {{- else -}} {{ with $caption }}{{ . }}{{ end }} {{- end -}} {{- end -}}
